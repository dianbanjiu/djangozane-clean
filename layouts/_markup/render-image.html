{{- $destination := .Destination -}} {{- $url := urls.Parse $destination -}} {{- $path := $url.Path -}} {{- $isRemote := or $url.Scheme (strings.HasPrefix $destination "//") -}} {{- $resource := false -}} {{- if and (not $isRemote) $path -}} {{- $resourcePath := strings.TrimPrefix "/" $path -}} {{- with .PageInner.Resources.GetMatch $path -}} {{- $resource = . -}} {{- else -}} {{- with .PageInner.Resources.GetMatch $resourcePath -}} {{- $resource = . -}} {{- else -}} {{- with resources.Get $resourcePath -}} {{- $resource = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $isProcessableImage := and $resource (reflect.IsImageResourceProcessable $resource) -}} {{- $processable := and $isProcessableImage (ne $resource.MediaType.SubType "gif") -}} {{- $sizes := "(min-width: 768px) 720px, calc(100vw - 2rem)" -}} {{- if $processable -}} {{- $displayWidth := $resource.Width -}} {{- if gt $displayWidth 1280 -}} {{- $displayWidth = 1280 -}} {{- end -}} {{- $candidateWidths := slice -}} {{- range slice 480 768 1024 1280 1600 -}} {{- if le . $resource.Width -}} {{- $candidateWidths = $candidateWidths | append . -}} {{- end -}} {{- end -}} {{- if not (in $candidateWidths $displayWidth) -}} {{- $candidateWidths = $candidateWidths | append $displayWidth -}} {{- end -}} {{- $fallback := $resource.Resize (printf "%dx q82" $displayWidth) -}} {{- $fallbackSet := slice -}} {{- $webpSet := slice -}} {{- range $candidateWidths -}} {{- $fallbackItem := $resource.Resize (printf "%dx q82" .) -}} {{- $webpItem := $resource.Resize (printf "%dx webp q78" .) -}} {{- $fallbackSet = $fallbackSet | append (printf "%s %dw" $fallbackItem.RelPermalink .) -}} {{- $webpSet = $webpSet | append (printf "%s %dw" $webpItem.RelPermalink .) -}} {{- end -}} {{- if .IsBlock -}} {{- else -}} {{- end -}} {{- else -}} {{- $src := $destination -}} {{- with $resource -}} {{- $src = .RelPermalink -}} {{- end -}} {{- if .IsBlock -}} {{- else -}} {{- end -}} {{- end -}} {{- /* chomp trailing newline */ -}}
